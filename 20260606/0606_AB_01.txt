public class BubbleSortEx {
    public static void BubbleSort(int[] arr) {
        int n = arr.Length;
        for (int i = 0; i < n - 1; i++) {
            for (int j = 0; j < n - 1 - i; j++) {
                if (arr[j] > arr[j + 1]) {
                    int temp = arr[j]; arr[j] = arr[j + 1]; arr[j + 1] = temp;
                }
            }
            System.Console.WriteLine((i + 1) + "周目: " + string.Join(",", arr));
        }
    }
    public static void Main() {
        int[] data = { 5, 3, 8, 4, 2 };
        System.Console.WriteLine("初期: " + string.Join(",", data));
        BubbleSort(data);
    }
}
