public class NewtonMethod {
    public static double F(double x) { return x * x - 2.0; }
    public static double Df(double x) { return (F(x + 1e-5) - F(x)) / 1e-5; }

    public static void Main() {
        double x = 3.0; // 初期値
        double epsilon = 1e-10; // 許容誤差
        while (System.Math.Abs(F(x)) > epsilon) {
            x = x - F(x) / Df(x); // ニュートン法の更新式
            System.Console.WriteLine("近似値: " + x);
        }
        System.Console.WriteLine("最終結果: " + x);
    }
}
