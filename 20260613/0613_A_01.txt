public class NumericalDiff {
    public static double F(double x) { return x * x - 2.0; }

    // 数値微分 f'(x)
    public static double Df(double x) {
        double h = 1e-5; // 0.00001
        return (F(x + h) - F(x)) / h;
    }

    public static void Main() {
        // x=3 のときの傾きは本来 2*x = 6
        System.Console.WriteLine("傾きの近似値: " + Df(3.0));
    }
}
